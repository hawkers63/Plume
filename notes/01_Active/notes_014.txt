1. EXECUTIVE SUMMARY
====================

Add a searchable, local French slang reference with British English meanings,
register labels and usage notes. Make it available while drafting the source and
while finalising a translated message. Keep Plume's primary translation intact:
any freely edited final message belongs in a clearly labelled local draft.

Do not put the complete attachment into the existing finishing-touch menus.
Its 195 table rows mix abbreviations, greetings, vocabulary, relationships,
reactions, idioms and complete utterances. Many are unsuitable as sentence
suffixes. Some alter an assertion, request or relationship. Others have ambiguous
or misleading glosses and need editorial verification before catalogue release.

Recommended user experience:
- One 'French slang…' button beside the input heading opens a reusable window.
- Search French terms, expansions and English meanings; filter by category.
- Each item shows its raw French form, meaning, register and placement guidance.
- 'Insert in source' inserts at the current caret only for a French/auto source.
- 'Copy term' copies only the raw French, without the English explanation.
- 'Insert in draft' works in a separate, editable French message preview.
- 'Copy draft' and 'Copy draft as HTML' copy that preview after the user reviews it.
- The reference makes no network requests; it does not automatically translate,
  send messages, add glossary rules or persist conversation text.

Plume now has the writing profile, tone-bearing presets, transactional preset
saves, Markdown diff and clipboard fixes discussed in notes_013. This report
updates that earlier assessment rather than proposing those features twice.
The existing input widget is self.input_box; it is not a persistence target.
Preset records belong in plume_config.json. There is no reason to introduce
scriptorium_config.json or an additional WRITING_PRESETS store.

Priority: catalogue semantics and deliberate draft composition first; native
drop research and streaming remain separate architectural work. A documented

def slang_search_key(value: str) -> str:
    """Normalise search only; preserve the original spelling for insertion."""
    value = value.casefold().replace("’", "'")
    decomposed = unicodedata.normalize("NFKD", value)
    return "".join(
        character for character in decomposed
        if not unicodedata.combining(character)
    )


def search_slang(query="", category="All", catalogue=SLANG_CATALOGUE):
    """Return matching records in catalogue order without a backend call."""
    words = slang_search_key(query).split()
    matches = []
    for record in catalogue:
        if category != "All" and record[4] != category:
            continue
        searchable = slang_search_key(" ".join(record[1:]))
        if all(word in searchable for word in words):
            matches.append(record)
    return matches


def slang_insertion(text: str, offset: int, term: str, limit: int):
    """Return new text and caret, adding spaces only at word boundaries."""
    if type(offset) is not int or not 0 <= offset <= len(text):
        raise ValueError("Choose a valid insertion position.")
    if not isinstance(term, str) or not term or "\x00" in term:
        raise ValueError("Choose a catalogue term.")
    left, right = text[:offset], text[offset:]
    before = " " if left and left[-1].isalnum() and term[0].isalnum() else ""
    after = " " if right and right[0].isalnum() and term[-1].isalnum() else ""
    inserted = before + term + after
    candidate = left + inserted + right
    if len(candidate) > limit:
        raise ValueError("The insertion would exceed the character limit.")
    return candidate, len(left) + len(inserted)


def find_slang_text_widget(widget):
    """Find Tk's public Text child without a CTk private attribute."""
    if isinstance(widget, tk.Text):
        return widget
    for child in widget.winfo_children():
        found = find_slang_text_widget(child)
        if found is not None:
            return found
    return None
```

def _open_slang_reference(self):
    """Reuse one local reference window without submitting any text."""
    existing = getattr(self, "_slang_reference", None)
    if existing is not None and existing.winfo_exists():
        existing.lift()
        existing.focus_set()
        return
    self._slang_reference = SlangReferenceDialog(self)


def _insert_slang_source(self, term):
    """Insert into a French source without changing the selected direction."""
    if self.direction_var.get() == DIR_EN_FR:
        raise ValueError("Choose a French source direction before inserting.")
    if self._import_busy or self.translate_btn.cget("state") == "disabled":
        raise ValueError("Wait for the current operation before inserting.")
    source = find_slang_text_widget(self.input_box)
    if source is None:
        raise ValueError("The source insertion point is unavailable.")
    text = self._input_text()
    # Python character length keeps offsets correct for non-BMP Unicode.
    offset = len(source.get("1.0", "insert"))
    limit = coerce_positive_int(
        self.config_data.get("max_input_chars"), DEFAULT_MAX_INPUT_CHARS,
    )
    candidate, caret = slang_insertion(text, offset, term, limit)
    fragment = candidate[offset:caret]
    source.insert("insert", fragment)
    self._request_id += 1
    self._clear_results()
    self._on_input_change()
    source.focus_set()
```

Offset calculation uses Python's length of the prefix rather than Tcl's count
of UTF-16 units. Insertion uses Tk's existing mark, avoiding numeric Tk indices
that might disagree for emoji. Source insertion rejects English-to-French mode;
Auto-detect is allowed without guessing or making a language-detection API call.
No pre-existing selected text is deleted. A message may need manual editing.

    def _refresh(self, _event=None):
        """Bound the number of widgets even when the catalogue grows."""
        for child in self.entries.winfo_children():
            child.destroy()
        matches = search_slang(self.query.get(), self.category.get())
        for index, record in enumerate(matches[:30]):
            _ident, term, expansion, meaning, category, register, note = record
            card = ctk.CTkFrame(self.entries)
            card.grid(row=index, column=0, sticky="ew", pady=4)
            card.grid_columnconfigure(0, weight=1)
            title = "{} — {}".format(term, meaning)
            detail = "{} · {}. {}".format(category, register, note)
            if expansion:
                detail = "Expanded: {}. ".format(expansion) + detail
            ctk.CTkLabel(
                card, text=title, wraplength=520, justify="left",
            ).grid(row=0, column=0, sticky="w", padx=8)
            ctk.CTkLabel(
                card, text=detail, wraplength=520, justify="left",
            ).grid(row=1, column=0, sticky="w", padx=8)
            row = ctk.CTkFrame(card, fg_color="transparent")
            row.grid(row=2, column=0, sticky="w", padx=8, pady=6)
            for column, (label, target) in enumerate((
                ("Copy term", "copy"), ("Insert in source", "source"),
                ("Insert in draft", "draft"),
            )):
                ctk.CTkButton(
                    row, text=label, width=130,
                    command=lambda t=term, dest=target: self._act(t, dest),
                ).grid(row=0, column=column, padx=(0, 6))
        if len(matches) > 30:
            self.notice.configure(
                text="Showing 30 matches; refine your search.",
            )
        else:
            self.notice.configure(
                text="{} matching terms.".format(len(matches)),
            )

    def _act(self, term, target):
        """Copy or insert only raw French; never the explanatory gloss."""
        try:
            if target == "copy":
                self.app._copy(term)
            elif target == "source":
                self.app._insert_slang_source(term)
            else:
                if self.draft_widget is None:
                    raise ValueError(
                        "The draft insertion point is unavailable."
                    )
                text = self.draft.get("1.0", "end-1c")
                offset = len(self.draft_widget.get("1.0", "insert"))
                candidate, caret = slang_insertion(text, offset, term, 10000)
                self.draft_widget.insert("insert", candidate[offset:caret])
                self.draft_widget.focus_set()
                self._update_metrics()
            self.notice.configure(
                text="Done. Review the wording before sending.",
            )
        except (ValueError, tk.TclError) as exc:
            message = (
                str(exc) if isinstance(exc, ValueError) else "Action failed."
            )
            self.notice.configure(text=message)

    def _draft_changed(self, _event=None):
        if self.draft_widget.edit_modified():
            self.draft_widget.edit_modified(False)
            self._update_metrics()

    def _update_metrics(self):
        self.metrics.configure(text=format_metrics_label(text_metrics(
            self.draft.get("1.0", "end-1c"), language=FRENCH,
        )))

    def _copy_draft(self, rich=False):
        """Use checked HTML transfer with the existing plain fallback."""
        text = self.draft.get("1.0", "end-1c")
        if not text:
            return
        if len(text) > 10000 or "\x00" in text:
            self.notice.configure(
                text="Use a draft of at most 10,000 characters.",
            )
            return
        fragment = "<p>{}</p>".format(html.escape(text).replace("\n", "<br>"))
        try:
            if not rich or not copy_html_to_windows_clipboard(
                fragment, text, self.winfo_id(),
            ):
                self.app._copy(text)
        except tk.TclError:
            self.notice.configure(
                text="The clipboard is busy. Please try again.",
            )
            return
        self.notice.configure(
            text="Draft copied. The translation has not been overwritten.",
        )
```

def validate_tones(value) -> list:
    """Keep the latest three compatible, unique catalogue tones."""
    if not isinstance(value, (list, tuple)):
        return []
    allowed = set(REGISTER_TONES) - {TONE_NONE}
    chosen = []
    for item in value:
        if not isinstance(item, str) or item not in allowed:
            continue
        chosen = [
            existing for existing in chosen
            if existing != item
            and frozenset({existing, item}) not in TONE_CONFLICTS
        ]
        chosen.append(item)
        chosen = chosen[-MAX_TONES:]
    return chosen
```

def normalise_slang_favourites(value, catalogue=SLANG_CATALOGUE):
    """Keep at most twenty known IDs in the user's chosen order."""
    if not isinstance(value, list):
        return []
    allowed = {record[0] for record in catalogue}
